Pole= < input size="5px" id="Pole" readonly>< input type="button" name="pole_loievskyi" value="Oblicz" Onclick="pole_loievskyi()"> < br>
obw= < input size="5px" id="obw" readonly>< input type="button" name="obw_loievskyi" value="Oblicz" Onclick="obw_loievskyi()">

< script type="text/javascript">
function pole_loievskyi() {
var a_loi= parseFloat(document.getElementById('dlugosc_a').value);
var b_loi= parseFloat(document.getElementById('dlugosc_b').value);
document.getElementById('Pole').value =(a_loi*b_loi).toFixed(2);
}
function obw_loievskyi() {
var a_loi= parseFloat(document.getElementById('dlugosc_a').value);
var b_loi= parseFloat(document.getElementById('dlugosc_b').value);
document.getElementById('obw').value =(a_loi*2+b_loi*2).toFixed(2);
}
