< script type="text/javascript">
function funkcja_loi() {
var d= parseFloat(document.getElementById('dlugosc_d').value);
var x= parseFloat(document.getElementById('dlugosc_x').value);
var drabina= Math.sqrt(d*d+x*x);
var alfa= Math.atan(d/x)*180/Math.PI;


document.getElementById('Drabina').value =drabina;
document.getElementById('Alfa').value =alfa.toFixed(2);
}
